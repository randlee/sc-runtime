// sc-storage; std-only, no sc-*, SQLx, rusqlite, ATM or sc-compose deps.
use std::{future::Future,pin::Pin};
pub enum TransactionMode { ReadOnly, ReadWrite }
pub struct TransactionSpec { pub name: &'static str, pub mode: TransactionMode }
pub type TransactionFuture<'tx,R> = Pin<Box<dyn Future<Output=Result<R,StorageError>>+Send+'tx>>;
pub trait TransactionProducer<H,R>: Send+Sync+'static {
fn start<'tx>(&'tx self, tx:&'tx mut H)->TransactionFuture<'tx,R>;
}
pub trait StorageRow: Send+Sync+'static {
type Key: Send+Sync+'static;
type New: Send+'static;
fn schema()-> &'static RowSchema;
fn encode(&self)->Result<Vec<Value>,StorageError>;
fn decode(values:Vec<Value>)->Result<Self,StorageError> where Self:Sized;
}
pub trait TransactionHandle: Send {
fn read<T:StorageRow>(&mut self,key:T::Key)->impl Future<Output=Result<Option<T>,StorageError>>+Send;
fn write<T:StorageRow>(&mut self,row:T)->impl Future<Output=Result<(),StorageError>>+Send;
}
// Optional CRUD primitives built for option (a), the proposed template extension,extension; notno domain SQL.
pub trait CrudTransaction: TransactionHandle {
fn insert<T:StorageRow>(&mut self,new:T::New)->impl Future<Output=Result<T,StorageError>>+Send;
fn delete<T:StorageRow>(&mut self,key:T::Key)->impl Future<Output=Result<bool,StorageError>>+Send;
}
pub trait AtomicStorage: Send+Sync {
type Transaction: TransactionHandle;
fn transaction<R:Send+'static,P:TransactionProducer<Self::Transaction,R>>(
&self,spec:TransactionSpec,producer:P
)->impl Future<Output=Result<R,StorageError>>+Send;
}
pub trait Storage<T>: AtomicStorage {
type Key: Send+Sync;
type Error: std::error::Error+Send+Sync+'static;
fn read(&self,key:Self::Key)->impl Future<Output=Result<Option<T>,Self::Error>>+Send;
fn read_all(&self)->impl Future<Output=Result<Vec<T>,Self::Error>>+Send;
fn write(&self,row:T)->impl Future<Output=Result<(),Self::Error>>+Send;
}
