Methodology

  • Timed iterations: 30 measured runs per case, each preceded by 1 untimed warmup (31 executions of the workload per case).
  • Wall clock: Rust Instant over the timed block only (warmup excluded).
  • Subprocess RSS: macOS /usr/bin/time -l on a separate one-shot command (isolates child peak).
  • Monty in-process RSS: getrusage(RUSAGE_SELF) after the timed Monty block—includes the test binary and Monty VM, not a standalone daemon.
  • Why Monty had no RSS earlier: the harness only attached time -l to subprocess cases; the Monty path runs inside cargo test until we added in-process getrusage and documented the compose child peak separately.

// production path: execute_feed_with_compose_program (not the lab feed_start harness)
let mut progress = match repl.feed_start(
    phase_snippet(phase),
    snippet_inputs(hook),
    PrintWriter::Disabled,
) {
    Ok(progress) => progress,
    Err(error) => return from_exception(error.error, stage),
};
loop {
    progress = match progress {
        ReplProgress::FunctionCall(call) if call.function_name == "render_template" => {
            match resume_render(call, stage, compose_program, host_deadline) {
                Ok(next) => next,
                Err(done) => return done,
            }
        }
        ReplProgress::FunctionCall(call) if call.function_name == "sc_compose_run" => {
            match resume_compose_run(call, stage, compose_program, host_deadline) {
                Ok(next) => next,
                Err(done) => return done,
            }
        }
        ReplProgress::Complete { repl, value } => {
            return finish_phase(phase, generation, repl, &value);
        }
        ReplProgress::FunctionCall(call) => {
            return denied(stage, &format!("host function {}", call.function_name));
        }
        ReplProgress::OsCall(call) => {
            let operation: &'static str = call.function_call.into();
            return denied_os(stage, operation);
        }
        ReplProgress::NameLookup(lookup) => return undefined_name(stage, &lookup.name),
        ReplProgress::ResolveFutures(_) => return denied(stage, "unresolved futures"),
    };
}
